cmake_minimum_required(VERSION 3.18)
project(kvaf_shared_ddr_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(DEFAULT_KVAF_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(KVAF_SOURCE_DIR "${DEFAULT_KVAF_SOURCE}" CACHE PATH "Matching KVAF checkout")
set(PROBE_ROLE "cpu" CACHE STRING "cpu, npu or host-test")
option(PROBE_EXTENDED_SGE "Use verified vendor 32-byte SGE instead of the pinned header layout" OFF)
if(PROBE_EXTENDED_SGE)
  add_compile_definitions(PROBE_EXTENDED_SGE=1)
endif()
set(UMDK_INCLUDE_DIR "" CACHE PATH "Directory containing target urma_api.h")
if(NOT EXISTS "${UMDK_INCLUDE_DIR}/urma_api.h")
  message(FATAL_ERROR "Set UMDK_INCLUDE_DIR to the target UMDK include directory")
endif()
include_directories("${UMDK_INCLUDE_DIR}" "${KVAF_SOURCE_DIR}/common/include"
  "${KVAF_SOURCE_DIR}/runtime/include" "${KVAF_SOURCE_DIR}/access_operator/include"
  "${KVAF_SOURCE_DIR}/meta_manager/include")
if(PROBE_ROLE STREQUAL "cpu")
  find_library(PROBE_URMA NAMES urma REQUIRED)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(KVAF_BUILD_REMOTE_CPU_SERVICE OFF CACHE BOOL "" FORCE)
  set(KVAF_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
  set(KVAF_ENABLE_AIV_BATCH_GET OFF CACHE BOOL "" FORCE)
  set(KVAF_ENABLE_AIV_BATCH_PUT OFF CACHE BOOL "" FORCE)
  add_subdirectory("${KVAF_SOURCE_DIR}" kvaf)
  add_executable(kvaf_ddr_probe k_service.cc)
  target_compile_definitions(kvaf_ddr_probe PRIVATE
    PROBE_CPU_SOURCE="${KVAF_SOURCE_DIR}/runtime/src/cpu_runtime.cc")
  target_link_libraries(kvaf_ddr_probe PRIVATE kvaf_runtime kvaf_peer_link "${PROBE_URMA}" dl pthread)
elseif(PROBE_ROLE STREQUAL "npu")
  set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_HOME_PATH}" CACHE PATH "CANN toolkit")
  add_library(kvaf_probe_hbm SHARED hbm_export.cc
    "${KVAF_SOURCE_DIR}/access_operator/src/direct_hccp/direct_hccp_session.cc"
    "${KVAF_SOURCE_DIR}/access_operator/src/direct_hccp/direct_hccp_remote_transport.cc"
    "${KVAF_SOURCE_DIR}/runtime/src/peer_link.cc")
  target_include_directories(kvaf_probe_hbm PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/include")
  target_link_directories(kvaf_probe_hbm PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/lib64")
  target_compile_definitions(kvaf_probe_hbm PRIVATE _GLIBCXX_USE_CXX11_ABI=0)
  target_link_libraries(kvaf_probe_hbm PRIVATE ascendcl dl pthread)
elseif(PROBE_ROLE STREQUAL "host-test")
  add_executable(kvaf_probe_host_test host_test.cc)
  target_compile_definitions(kvaf_probe_host_test PRIVATE
    PROBE_CPU_SOURCE="${KVAF_SOURCE_DIR}/runtime/src/cpu_runtime.cc")
  target_compile_options(kvaf_probe_host_test PRIVATE -O1 -ffunction-sections -fdata-sections)
  if(APPLE)
    # Only the fake-provider test runs on macOS; it never calls madvise.
    target_compile_definitions(kvaf_probe_host_test PRIVATE MADV_HUGEPAGE=14)
    target_link_options(kvaf_probe_host_test PRIVATE -Wl,-dead_strip)
  else()
    target_link_options(kvaf_probe_host_test PRIVATE -Wl,--gc-sections)
    target_link_libraries(kvaf_probe_host_test PRIVATE dl pthread)
  endif()
else()
  message(FATAL_ERROR "PROBE_ROLE must be cpu, npu or host-test")
endif()
